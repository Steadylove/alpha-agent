// ==================== 3. 单标的形态识别与全周期收益核算 ====================
f_eval_stock(int s_year, bool filter_rsi, int rsi_th) =>
    dif = (ta.ema(close, 12) - ta.ema(close, 26)) * 100.0
    dea = ta.ema(dif, 9)
    macd = (dif - dea) * 2.0
    
    tp_p = math.max(close, open)
    llv_p = ta.lowest(tp_p, 4)
    tp_dea_u = dea > dea[1] and dea[2] > dea[1]
    tpdu_d = ta.barssince(tp_dea_u[1])
    dc = ta.cross(dea, dif) and dea[1] < dif[1]
    dc_d1 = ta.barssince(dc)
    cond_macd_n = macd < 0 and macd[1] < 0
    
    lv_c1 = ta.lowest(close, nz(dc_d1) + 1)
    lv_c2 = lv_c1[dc_d1 + 1]
    lv_c3 = lv_c2[dc_d1 + 1]
    
    lv_d1 = ta.lowest(dif, nz(dc_d1) + 1)
    lv_d2 = lv_d1[dc_d1 + 1]
    lv_d3 = lv_d2[dc_d1 + 1]
    
    log_ld2 = math.floor(math.log10(math.abs(lv_d2))) - 1
    log_ld3 = math.floor(math.log10(math.abs(lv_d3))) - 1
    
    diff_dif_ld2 = math.floor(dif / math.pow(10, log_ld2))
    diff_dif_ld3 = math.floor(dif / math.pow(10, log_ld3))
    diff_llv_ld2 = math.floor(lv_d2 / math.pow(10, log_ld2))
    diff_llv_ld3 = math.floor(lv_d3 / math.pow(10, log_ld3))
    
    btm_d_direct = cond_macd_n and diff_dif_ld2 > diff_llv_ld2 and diff_dif_ld2 <= diff_dif_ld2[1] and lv_c1 < lv_c2
    btm_d_indirect = cond_macd_n and diff_dif_ld3 > diff_llv_ld3 and diff_dif_ld3 <= diff_dif_ld3[1] and lv_c1 < lv_c3 and lv_c3 < lv_c2
    
    cond_buy = (diff_dif_ld2 > diff_dif_ld2[1] and btm_d_direct[1]) or (diff_dif_ld3 > diff_dif_ld3[1] and btm_d_indirect[1])
    cond_btm_psvn = btm_d_direct or btm_d_indirect
    btm_d_dis = (btm_d_direct[1] and lv_d1 <= lv_d2) or (btm_d_indirect[1] and lv_d1 <= lv_d3)
    
    cond_psvn_dis = (ta.cross(dif, dea) and dif[1] < dea[1]) and ta.barssince(btm_d_dis) < ta.barssince(cond_btm_psvn) and ta.barssince(cond_btm_psvn) < dc_d1
    cond_buy1 = cond_psvn_dis and (ta.ema(high, 24) < ta.ema(low, 90) or ta.ema(high, 24)[1] < ta.ema(low, 90)[1])
    
    buy1_sig = ta.barssince(cond_buy[1]) > 10 and cond_buy
    buy2_sig = ta.barssince(cond_buy1[1]) > 10 and cond_buy1
    
    // 相对强度 (RS 1~99 动能评分)
    p21  = (not na(close[21])  and close[21]  != 0) ? (close - close[21])  / close[21]  : 0.0
    p63  = (not na(close[63])  and close[63]  != 0) ? (close - close[63])  / close[63]  : 0.0
    p126 = (not na(close[126]) and close[126] != 0) ? (close - close[126]) / close[126] : 0.0
    p252 = (not na(close[252]) and close[252] != 0) ? (close - close[252]) / close[252] : 0.0
    weighted_perf = (0.20 * p21 + 0.40 * p63 + 0.20 * p126 + 0.20 * p252) * 100.0
    float rs_rating = math.min(99.0, math.max(1.0, 50.0 + 48.0 * (weighted_perf / (math.abs(weighted_perf) + 28.0))))

    // RSI 过滤判定
    atr_risk = ta.sma(ta.atr(14), 14)
    rsi_val  = ta.sma(ta.rsi(close, 14), 14)
    bool pass_rsi = filter_rsi ? (rsi_val > rsi_th) : true
    
    var float entry_p         = na
    var int   sig_type        = 0
    var float sl_level        = na
    var float tp_level        = na
    var float hi_p            = na
    var float max_pnl         = 0.0
    
    var float ytd_closed_pnl  = 0.0
    var int   ytd_trade_cnt   = 0
    var int   ytd_win_cnt     = 0
    
    bool is_ytd_bar = year(time) >= s_year
    
    // 开仓点火
    if buy1_sig and sig_type == 0 and pass_rsi
        entry_p  := close
        sig_type := 1
        sl_level := close - 4.0 * atr_risk
        tp_level := close - 5.5 * atr_risk
        hi_p     := high
        max_pnl  := 0.0
    else if buy2_sig and sig_type == 0 and pass_rsi
        entry_p  := close
        sig_type := 2
        sl_level := close - 4.0 * atr_risk
        tp_level := close - 5.5 * atr_risk
        hi_p     := high
        max_pnl  := 0.0
        
    // 动态风控与平仓清算
    if sig_type > 0
        hi_p    := math.max(hi_p, high)
        max_pnl := math.max(max_pnl, (close - entry_p) / entry_p * 100.0)
        
        float cur_mult = max_pnl >= 50.0 ? 3.5 : (max_pnl >= 25.0 ? 4.2 : 5.5)
        tp_level := math.max(tp_level, hi_p - (cur_mult * atr_risk))
        
        if max_pnl >= 10.0
            sl_level := math.max(sl_level, entry_p * 1.01) // 保本锁定
            
        // 平仓结算：累加已实现损益与胜率账本
        if close < sl_level or close < tp_level
            float closed_pnl = (close - entry_p) / entry_p * 100.0
            if is_ytd_bar
                ytd_closed_pnl := ytd_closed_pnl + closed_pnl
                ytd_trade_cnt  := ytd_trade_cnt + 1
                if closed_pnl > 0
                    ytd_win_cnt := ytd_win_cnt + 1
            sig_type := 0
            entry_p  := na
            sl_level := na
            tp_level := na
            hi_p     := na
            max_pnl  := 0.0

    // 单票综合收益：已平仓兑现 + 当前浮动
    float cur_float_pnl = sig_type > 0 ? (close - entry_p) / entry_p * 100.0 : 0.0
    float total_ytd_pnl = ytd_closed_pnl + (sig_type > 0 ? cur_float_pnl : 0.0)
    
    [sig_type, entry_p, close, cur_float_pnl, sl_level, total_ytd_pnl, ytd_closed_pnl, rs_rating, ytd_trade_cnt, ytd_win_cnt]

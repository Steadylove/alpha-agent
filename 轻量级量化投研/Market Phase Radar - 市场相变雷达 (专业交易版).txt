    nz(ta.percentrank(nz(src, 0.0), len), 50.0)

    float rating = score >= 195.93 ? 99.0 :
                   score >= 117.11 ? 90.0 + (score - 117.11) / (195.93 - 117.11) * 8.9 :
                   score >= 99.04  ? 70.0 + (score - 99.04) / (117.11 - 99.04) * 19.9 :
                   score >= 91.66  ? 50.0 + (score - 91.66) / (99.04 - 91.66) * 19.9 :
                   score >= 80.96  ? 30.0 + (score - 80.96) / (91.66 - 80.96) * 19.9 :
                   score >= 53.64  ? 10.0 + (score - 53.64) / (80.96 - 53.64) * 19.9 :
                   score >= 24.86  ? 2.0 + (score - 24.86) / (53.64 - 24.86) * 7.9 : 1.0
    math.max(1.0, math.min(99.0, rating))

// 4. MPR 状态拓扑与传导决策双层引擎 (与 MarketCompass 100% 镜像)
// =============================================================================
f_calc_mpr_v2_core(int c_len, int d_len, float t_persist) =>
    vix_c_loc   = request.security("CBOE:VIX", "D", close, ignore_invalid_symbol=true)
    vix9d_c_loc = request.security("CBOE:VIX9D", "D", close, ignore_invalid_symbol=true)
    vix3m_c_loc = request.security("CBOE:VIX3M", "D", close, ignore_invalid_symbol=true)
    spy_c_loc   = request.security("AMEX:SPY", "D", close, ignore_invalid_symbol=true)
    spy_vol_loc = request.security("AMEX:SPY", "D", volume, ignore_invalid_symbol=true)
    rsp_c_loc   = request.security("AMEX:RSP", "D", close, ignore_invalid_symbol=true)
    tlt_c_loc   = request.security("AMEX:TLT", "D", close, ignore_invalid_symbol=true)
    dxy_c_loc   = request.security("TVC:DXY", "D", close, ignore_invalid_symbol=true)
    gld_c_loc   = request.security("AMEX:GLD", "D", close, ignore_invalid_symbol=true)
    hyg_c_loc   = request.security("AMEX:HYG", "D", close, ignore_invalid_symbol=true)
    iei_c_loc   = request.security("AMEX:IEI", "D", close, ignore_invalid_symbol=true)

    // SPY 日线均线基准 (绝对趋势闸门)
    spy_ema20 = ta.ema(spy_c_loc, 20)
    spy_ema50 = ta.ema(spy_c_loc, 50)
    spy_ret5  = ta.roc(spy_c_loc, d_len)

    // F1: 量价推进效率
    spy_high_252 = ta.highest(spy_c_loc, c_len), spy_low_252 = ta.lowest(spy_c_loc, c_len)
    spy_loc      = (spy_high_252 - spy_low_252 == 0) ? 0.5 : (spy_c_loc - spy_low_252) / (spy_high_252 - spy_low_252)
    spy_rvol     = spy_vol_loc / nz(ta.sma(spy_vol_loc, 20), spy_vol_loc)
    spy_move     = math.abs(spy_c_loc - nz(spy_c_loc[d_len], spy_c_loc)) / nz(spy_c_loc[d_len], spy_c_loc) * 100.0
    spy_eff      = spy_move / (spy_rvol + 0.1)
    f1_raw       = spy_loc * math.log(1.0 + 1.0 / (spy_eff + 0.05))
    f1_stress    = f_rolling_ecdf(f1_raw, c_len)

    // F2: 隐波期限倒挂 (真实阈值过滤)
    raw_term     = (not na(vix3m_c_loc) and vix3m_c_loc > 0) ? (nz(vix9d_c_loc, vix_c_loc) / vix3m_c_loc) : 1.0
    f2_ecdf      = f_rolling_ecdf(raw_term, c_len)
    f2_stress    = (raw_term < 0.90 or vix_c_loc < 16.0) ? math.min(f2_ecdf, 45.0) : f2_ecdf

    // F3: 跨资产避险脱节
    tlt_ret      = ta.roc(tlt_c_loc, d_len)
    dxy_ret      = ta.roc(dxy_c_loc, d_len)
    macro_break  = (spy_ret5 < 0 ? 1.0 : 0.2) * (math.max(0.0, tlt_ret) + math.max(0.0, dxy_ret))
    f3_stress    = f_rolling_ecdf(macro_break, c_len)

    // F4: 信用利差紧缩
    raw_cred     = (not na(hyg_c_loc) and hyg_c_loc > 0) ? (nz(iei_c_loc, 115.0) / hyg_c_loc) : 1.0
    f4_stress    = f_rolling_ecdf(raw_cred, c_len)

    // F5: 现货广度背离
    rsp_ret      = ta.roc(rsp_c_loc, d_len)
    dispersion   = spy_ret5 - rsp_ret
    f5_stress    = f_rolling_ecdf(dispersion, c_len)

    // 三大物理域聚合
    float dom_vol_s   = f2_stress
    float dom_cred_s  = math.max(f4_stress, f3_stress)
    float dom_spot_s  = math.max(f5_stress, f1_stress)

    int sigma_vol   = dom_vol_s >= 75.0 ? 2 : (dom_vol_s >= 50.0 ? 1 : 0)
    int sigma_cred  = dom_cred_s >= 75.0 ? 2 : (dom_cred_s >= 50.0 ? 1 : 0)
    int sigma_spot  = dom_spot_s >= 75.0 ? 2 : (dom_spot_s >= 50.0 ? 1 : 0)

    // 几何耦合度
    int n_active = (sigma_vol >= 1 ? 1 : 0) + (sigma_cred >= 1 ? 1 : 0) + (sigma_spot >= 1 ? 1 : 0)
    float geom_tension = math.pow((dom_vol_s / 100.0) * (dom_cred_s / 100.0) * (dom_spot_s / 100.0), 1.0 / 3.0)
    float coupling_ratio = (n_active / 3.0) * geom_tension

    // 现货真实破坏度
    float spy_high_20 = ta.highest(spy_c_loc, 20)
    float spy_dd_pct  = (spy_high_20 > 0) ? (spy_c_loc - spy_high_20) / spy_high_20 * 100.0 : 0.0
    bool  is_spy_below_ema50 = spy_c_loc < spy_ema50

    float spy_damage = 0.0
    if spy_dd_pct <= -5.0 or (spy_dd_pct <= -2.5 and is_spy_below_ema50)
        spy_damage := 85.0
    else if spy_dd_pct <= -2.0 or is_spy_below_ema50
        spy_damage := 60.0
    else if spy_dd_pct <= -0.8
        spy_damage := 35.0
    else
        spy_damage := 10.0

    float lead_gap = math.max(dom_vol_s, dom_cred_s) - spy_damage
    var int lead_persist = 0
    lead_persist := (lead_gap > 20.0 and spy_damage < 40.0) ? (lead_persist + 1) : 0
    float lead_decay   = 1.0 - math.exp(-lead_persist / t_persist)
    float lead_quality = math.max(0.0, lead_gap * lead_decay)

    // 传导拓扑路径树
    int path_id = 0
    int trans_depth = 0
    if is_spy_below_ema50 and spy_damage >= 70.0
        path_id := 4
        trans_depth := 3
    else if (sigma_vol >= 1 or sigma_cred >= 1) and sigma_spot >= 1 and spy_damage < 60.0
        path_id := 2
        trans_depth := 2
    else if (sigma_vol >= 1 or sigma_cred >= 1) and sigma_spot == 0
        path_id := 1
        trans_depth := 1
    else if sigma_spot >= 1 and sigma_vol == 0 and sigma_cred == 0
        path_id := 3
        trans_depth := 1
    else
        path_id := 0
        trans_depth := 0

    // 状态映射条件概率
    float alpha_base = path_id == 4 ? 1.20 : (path_id == 2 ? 0.35 : (path_id == 1 ? -0.65 : (path_id == 3 ? -0.50 : -1.60)))
    float p_5d_logit = alpha_base + (0.90 * coupling_ratio) + (0.35 * math.log(lead_persist + 1.0))
    float prob_5d_down = (1.0 / (1.0 + math.exp(-p_5d_logit))) * 100.0

    float exp_severity = path_id == 4 ? 1.25 : (path_id == 2 ? 1.05 : (path_id == 1 ? 0.75 : (path_id == 3 ? 0.60 : 0.35)))
    float raw_risk_score = prob_5d_down * exp_severity * (1.0 + 0.15 * trans_depth)
    float market_risk_score = math.min(100.0, math.max(5.0, raw_risk_score))

    // 隐状态估计
    int fsm_state = path_id == 4 ? 3 : (path_id == 2 ? 2 : (path_id == 1 or path_id == 3 ? 1 : 0))

    // 动力学速度
    float trans_stress_agg = (dom_vol_s * 0.35 + dom_cred_s * 0.35 + dom_spot_s * 0.30)
    float trans_vel = (trans_stress_agg - nz(trans_stress_agg[dyn_len], trans_stress_agg)) / 100.0

    [fsm_state, path_id, spy_damage, lead_quality, f1_stress, f2_stress, f3_stress, f4_stress, f5_stress, raw_term, raw_cred, prob_5d_down, market_risk_score, trans_vel, lead_persist, lead_gap]

// ⚡ 跨周期日线基准执行 (彻底杜绝主图切换 4h/1h 时的差异)
